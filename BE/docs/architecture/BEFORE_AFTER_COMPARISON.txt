AFTER (Redis-Backed Sessions):
────────────────────────────────────────────────────────────────────────────
require('dotenv').config();
const express = require('express');
const session = require('express-session');
const RedisStore = require('connect-redis').default;
const redisClient = require('./redis/client');
const passport = require('./middleware/auth.middleware');
const authRouter = require('./routes/authRouter');
const projectRouter = require('./routes/projectRouter');
const startServer = require('./utils/startServer');

const app = express();

// Middleware
app.use(express.json());
app.use(express.urlencoded({ extended: true }));

// Session configuration with Redis store
// Falls back to memory store if Redis is unavailable (development)
const sessionConfig = {
  secret: process.env.SESSION_SECRET || 'change_this_secret',
  resave: false,
  saveUninitialized: false,
  cookie: {
    secure: process.env.NODE_ENV === 'production', // HTTPS only in production
    httpOnly: true, // Prevent XSS attacks
    sameSite: 'strict', // CSRF protection
    maxAge: 24 * 60 * 60 * 1000 // 24 hours
  }
};

// Add Redis store if client is connected
if (redisClient.isOpen || redisClient.isReady) {
  sessionConfig.store = new RedisStore({ client: redisClient });
  console.log('Session store: Redis');
} else {
  console.warn('Session store: In-memory (Redis not available)');
}

app.use(session(sessionConfig));

app.use(passport.initialize());
app.use(passport.session());

app.use('/api/auth', authRouter);
app.use('/api/projects', projectRouter);

// ...

IMPROVEMENTS:
  ✅ Sessions stored in Redis (persistent database)
  ✅ Sessions survive server restarts
  ✅ Multiple servers share same Redis (distributed)
  ✅ Memory managed by Redis (eviction policies)
  ✅ Secure cookie flags (XSS, CSRF protection)
  ✅ Graceful fallback to in-memory if Redis unavailable
  ✅ 24-hour session expiration

const { createClient } = require('redis');

// Initialize the client
const client = createClient({
    url: process.env.REDIS_URL

// Event listeners for production monitoring

LIMITATIONS:
  ❌ Only supports REDIS_URL (not separate host/port)
  ❌ No reconnection strategy
  ❌ Synchronous client creation (not connected yet)
  ❌ Basic shutdown error handling
  ❌ No support for environment variable variations


AFTER (Enhanced Redis Connection):
────────────────────────────────────────────────────────────────────────────
const { createClient } = require('redis');

// Initialize the client
// Supports both legacy REDIS_URL and separate REDIS_HOST/REDIS_PORT
const redisConfig = process.env.REDIS_URL 
  ? { url: process.env.REDIS_URL }
  : {
      host: process.env.REDIS_HOST || 'localhost',
      port: process.env.REDIS_PORT || 6379,
      password: process.env.REDIS_PASSWORD || undefined,
      socket: {
        reconnectStrategy: (retries) => Math.min(retries * 50, 500)
      }
    };

const client = createClient(redisConfig);

// Event listeners for production monitoring
client.on('connect', () => console.log('Redis: Connected'));
client.on('ready', () => console.log('Redis: Ready to use'));
client.on('error', (err) => console.error('Redis: Connection Error', err));
client.on('end', () => console.log('Redis: Connection closed'));

// Connect to Redis (non-blocking)
client.connect().catch((err) => {
  console.error('Failed to connect to Redis:', err.message);
  console.warn('Sessions will use in-memory store (development mode)');
});

// Handle graceful shutdown
process.on('SIGINT', async () => {
  try {
    await client.quit();
  } catch (err) {
    console.error('Error closing Redis connection:', err);
  }
  process.exit();
});

module.exports = client;

IMPROVEMENTS:
  ✅ Supports multiple configuration methods:
     - REDIS_URL (cloud providers)
     - REDIS_HOST/REDIS_PORT (local/Docker)
  ✅ Automatic reconnection strategy (exponential backoff)
  ✅ Connects asynchronously (non-blocking)
  ✅ Better error handling and fallback messages
  ✅ Graceful error handling on shutdown
  ✅ Ready for both development and production
